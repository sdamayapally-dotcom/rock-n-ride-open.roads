function step(dt){
  const c=CARS[S.car],sh=K.ShiftLeft||K.ShiftRight,nit=sh&&P.n>0,top=c.top/3.6*(nit?1.3:1);
  if(K.ArrowUp||nit)P.v+=c.acc*(nit?2:1)*Math.max(0,1-P.v/top)*dt;
  else if(!K.ArrowDown&&!K.Space)P.v-=(2.5+P.v*.03)*dt;
  if(P.v>top)P.v-=12*dt;
  if(K.ArrowDown)P.v-=28*dt;if(K.Space)P.v-=60*dt;
  P.v=Math.max(0,P.v);
  P.n=nit?Math.max(0,P.n-28*dt):Math.min(100,P.n+7*dt);
  const st=(K.ArrowRight?1:0)-(K.ArrowLeft?1:0),lim=RW/2-1.3;
  P.x+=st*(5+P.v*.12)*c.hd*Math.min(1,P.v/5)*dt;
  if(Math.abs(P.x)>lim){P.x=Math.sign(P.x)*lim;P.v*=1-.6*dt}
  P.z-=P.v*dt;P.d+=P.v*dt;P.spin=(P.spin||0)*Math.pow(.02,dt);P.yaw+=(-st*.14-P.yaw)*Math.min(1,8*dt);
  const cnt=[8,14,22][S.traf];
  TR.forEach((t,i)=>{
    t.m.visible=i<cnt;if(i>=cnt)return;
    t.z-=t.v*dt;
    if(t.z>P.z+50)place(t);else if(t.z<P.z-520)t.z=P.z+45+Math.random()*30;
    t.m.position.set(LXn(t.lane,t.typ),0,t.z);
    const dz=t.z-P.z,dx=P.x-LXn(t.lane,t.typ),h=2+t.m.userData.L/2;
    if(Math.abs(dz)<h&&Math.abs(dx)<1.9){
      const side=dx>=0?1:-1;
      if(dz<0&&Math.abs(dx)<1.2)P.v=Math.min(P.v,t.v*.6);
      else{P.x+=side*.6;P.v*=.85}
      P.spin=(P.spin||0)+side*.45+(Math.random()-.5)*.2;
      hitT=.3;$('fl').style.opacity=1;setTimeout(()=>$('fl').style.opacity=0,150);
    }
  });
  COINS.forEach(c=>{
    c.m.position.set(LXn(c.lane,c.typ),1,c.z);c.m.rotation.x+=dt*.15;
    if(!c.got&&Math.abs(c.z-P.z)<2.4&&Math.abs(LXn(c.lane,c.typ)-P.x)<1.7){c.got=1;c.m.visible=false;runCoins++;S.coins++;save()}
    if(c.z>P.z+30)placeCoin(c);
  });
  if(pBustT>0){
    pBustT-=dt;P.v=Math.min(P.v,2.5);
  }else{
    const heat=Math.max(0,P.v/(c.top/3.6)-0.55)*2.2;
    const r=policeThink(dt,heat);
    if(r=='busted'){
      const pen=Math.min(runCoins,6+Math.floor(Math.random()*7));
      runCoins-=pen;S.coins=Math.max(0,S.coins-pen);save();
      pBustT=2.4;policeReset();pFreeT=11+Math.random()*7;
      $('chase').style.display='block';$('chase').style.color='#ff2e44';$('chase').textContent='🚔 BUSTED! -'+pen+' 🪙';
      setTimeout(()=>{if(pBustT<=0)$('chase').style.display='none'},2400);
    }
  }
  pm.userData.f.visible=nit&&P.v>2;
  const u=S.mph?2.237:3.6;
  $('spd').textContent=Math.round(P.v*u);$('dist').textContent=(P.d/1000).toFixed(2)+' km';
  $('nb').style.width=P.n+'%';$('coins').textContent='🪙 '+runCoins;
}

let AC,mg,nb,st16=0,nT=0;
const BASS=[45,45,48,43],ARP=[[57,60,64,60],[57,60,64,60],[60,64,67,64],[55,59,62,59]],mf=n=>440*Math.pow(2,(n-69)/12);
const mvol=()=>{if(mg)mg.gain.value=[0,.1,.22][S.mus]};
function tone(f,t,d,ty,v){const o=AC.createOscillator(),g=AC.createGain();o.type=ty;o.frequency.value=f;g.gain.setValueAtTime(v,t);g.gain.exponentialRampToValueAtTime(.001,t+d);o.connect(g);g.connect(mg);o.start(t);o.stop(t+d)}
function kick(t){const o=AC.createOscillator(),g=AC.createGain();o.frequency.setValueAtTime(150,t);o.frequency.exponentialRampToValueAtTime(45,t+.12);g.gain.setValueAtTime(.9,t);g.gain.exponentialRampToValueAtTime(.001,t+.25);o.connect(g);g.connect(mg);o.start(t);o.stop(t+.25)}
function hat(t){const s=AC.createBufferSource(),f=AC.createBiquadFilter(),g=AC.createGain();s.buffer=nb;f.type='highpass';f.frequency.value=7000;g.gain.setValueAtTime(.25,t);g.gain.exponentialRampToValueAtTime(.001,t+.05);s.connect(f);f.connect(g);g.connect(mg);s.start(t);s.stop(t+.06)}
function sched(){const sp=60/112/4;while(nT<AC.currentTime+.2){
  const s=st16%16,b=(st16>>4)%4;
  if(s%4==0)kick(nT);if(s%4==2)hat(nT);
  if(s%2==0)tone(mf(BASS[b]),nT,sp*1.8,'sawtooth',.3);
  tone(mf(ARP[b][s%4]),nT,sp*.9,'square',.1);
  if(s==0)[0,1,2].forEach(k=>tone(mf(ARP[b][k]+12),nT,sp*15,'triangle',.05));
  st16++;nT+=sp}}
function musicStart(){
  if(!AC){AC=new(window.AudioContext||window.webkitAudioContext)();mg=AC.createGain();mg.connect(AC.destination);
    nb=AC.createBuffer(1,AC.sampleRate*.1,AC.sampleRate);const d=nb.getChannelData(0);for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
    nT=AC.currentTime+.1;setInterval(sched,50)}
  if(AC.state=='suspended')AC.resume();mvol();
}
